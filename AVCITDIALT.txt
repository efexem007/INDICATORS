f_rankLong(_priceLine, _signalLine, _mbl) =>
    int out = 0
    if _signalLine > _mbl
        out := 1
    else if _priceLine > _mbl and _signalLine < _mbl
        out := 2
    else if _priceLine < _mbl and _signalLine < _mbl
        out := 3
    out

f_rankShort(_priceLine, _signalLine, _mbl) =>
    int out = 0
    if _signalLine < _mbl
        out := 1
    else if _priceLine < _mbl and _signalLine > _mbl
        out := 2
    else if _priceLine > _mbl and _signalLine > _mbl
        out := 3
    out

f_emaSeed(_src, _length) =>
    float alpha = 2.0 / (_length + 1.0)
    float seed = ta.sma(_src, _length)
    var float out = na
    if bar_index < _length - 1
        out := na
    else if bar_index == _length - 1
        out := seed
    else
        out := _src * alpha + nz(out[1], seed) * (1.0 - alpha)
    out

f_rollingPercentile(_src, _length) =>
    float current = _src
    int valid = 0
    int belowOrEqual = 0
    if not na(current)
        for i = 0 to _length - 1
            float v = _src[i]
            if not na(v)
                valid += 1
                if v <= current
                    belowOrEqual += 1
    float out = na
    if valid >= math.min(20, _length)
        out := float(belowOrEqual) / float(valid)
    out

// ============================================================================
// HELPERS - FATIGUE
// ============================================================================
f_rankOf(_src, _count) =>
    float x = _src
    int less = 0
    int eq = 0
    if not na(x) and _count > 0
        for i = 0 to _count - 1
            float v = _src[i]
            if not na(v)
                if v < x
                    less += 1
                else if v == x
                    eq += 1
    float result = na
    if not na(x) and _count > 0
        result := (less + (eq + 1) / 2.0) / _count * 100.0
    result

// ENGINE A - TDIALT   (yalnizca aktif grafik)

f_core() =>
    float haClose = (open + high + low + close) / 4.0
    var float haOpen = na
    if na(haOpen[1])
        haOpen := (open + close) / 2.0
    else
        haOpen := (haOpen[1] + haClose[1]) / 2.0
    float haHigh = math.max(high, math.max(haOpen, haClose))
    float haLow = math.min(low, math.min(haOpen, haClose))

    float srcOpen = useHeikinAshi ? haOpen : open
    float srcHigh = useHeikinAshi ? haHigh : high
    float srcLow = useHeikinAshi ? haLow : low
    float srcClose = useHeikinAshi ? haClose : close

    float rsi = ta.rsi(srcClose, rsiLength)

    float marketBaseLine = ta.sma(rsi, bandLength)
    float deviation = ta.stdev(rsi, bandLength, true)
    float upperBand = not na(marketBaseLine) and not na(deviation) ? marketBaseLine + BAND_MULTIPLIER * deviation : na
    float lowerBand = not na(marketBaseLine) and not na(deviation) ? marketBaseLine - BAND_MULTIPLIER * deviation : na
    float priceLine = ta.sma(rsi, priceLineLength)
    float signalLine = ta.sma(rsi, signalLineLength)

    float pacUpper = ta.rma(srcHigh, pacLength)
    float pacLower = ta.rma(srcLow, pacLength)

    float prevClose = srcClose[1]
    float trueRange = na(prevClose) ? srcHigh - srcLow : math.max(srcHigh - srcLow, math.max(math.abs(srcHigh - prevClose), math.abs(srcLow - prevClose)))
    float atrValue = ta.rma(trueRange, ATR_LENGTH)

    int effectiveFast = math.min(macdFast, macdSlow - 1)
    float emaFast = f_emaSeed(srcClose, effectiveFast)
    float emaSlow = f_emaSeed(srcClose, macdSlow)
    float macdLine = not na(emaFast) and not na(emaSlow) ? emaFast - emaSlow : na
    bool macdLongPass = false
    if not na(macdLine) and not na(macdLine[1])
        bool rising = macdLine > macdLine[1]
        bool falling = macdLine < macdLine[1]
        macdLongPass := macdLine > 0.0 ? rising : not falling

    float bandWidthPct = not na(marketBaseLine) and not na(deviation) ? (2.0 * BAND_MULTIPLIER * deviation) / math.max(math.abs(marketBaseLine), 0.000000001) : na
    bool inCalc = not limitCalc or bar_index >= last_bar_index - calcBars
    float bandWidthPercentile = na
    if inCalc
        bandWidthPercentile := f_rollingPercentile(bandWidthPct, REGIME_WINDOW)
    int regimeCode = 0
    if not na(bandWidthPercentile)
        if bandWidthPercentile <= 0.20
            regimeCode := 1
        else if bandWidthPercentile >= 0.90
            regimeCode := 3
        else
            regimeCode := 2

    bool coreReady = not na(priceLine) and not na(signalLine) and not na(marketBaseLine) and not na(pacUpper) and not na(pacLower)
    bool macdLongReady = not useMacdFilter or macdLongPass
    bool macdShortReady = not useMacdFilter or (not na(macdLine) and bar_index > 0 and not macdLongPass)

    bool longCommon = coreReady and macdLongReady and priceLine > signalLine and priceLine > oversold and priceLine < overbought and srcHigh > pacUpper and srcClose > srcOpen
    bool shortCommon = coreReady and macdShortReady and priceLine < signalLine and priceLine > oversold and priceLine < overbought and srcLow < pacLower and srcClose < srcOpen

    int longStrength = longCommon ? f_rankLong(priceLine, signalLine, marketBaseLine) : 0
    int shortStrength = shortCommon ? f_rankShort(priceLine, signalLine, marketBaseLine) : 0

    bool eligibleLong = longStrength > 0 and longStrength <= strengthLevel
    bool eligibleShort = shortStrength > 0 and shortStrength <= strengthLevel

    int confirmationSignedStrength = eligibleLong ? longStrength : eligibleShort ? -shortStrength : 0

    var int previousLong = 0
    var int previousShort = 0
    int alertSignedStrength = 0
    if eligibleLong and longStrength != previousLong
        alertSignedStrength := longStrength
    else if eligibleShort and shortStrength != previousShort
        alertSignedStrength := -shortStrength
    previousLong := longStrength
    previousShort := shortStrength

    float pacMid = not na(pacUpper) and not na(pacLower) ? (pacUpper + pacLower) / 2.0 : na
    float pacDistanceAtr = not na(pacMid) and not na(atrValue) and atrValue > 0.0 ? math.abs(srcClose - pacMid) / atrValue : na
    float pacWidthAtr = not na(pacUpper) and not na(pacLower) and not na(atrValue) and atrValue > 0.0 ? (pacUpper - pacLower) / atrValue : na

    bool chaseRisk = alertSignedStrength != 0 and not na(pacDistanceAtr) and pacDistanceAtr > 1.5
    bool widePacRisk = not na(pacWidthAtr) and pacWidthAtr > 1.4
    int riskCode = 0
    if chaseRisk
        riskCode += 1
    if widePacRisk
        riskCode += 2
    if regimeCode == 3
        riskCode += 4
    if regimeCode == 1
        riskCode += 8

    float bandPosition = not na(priceLine) and not na(lowerBand) and not na(upperBand) and upperBand > lowerBand ? (priceLine - lowerBand) / (upperBand - lowerBand) : na

    bool priceSignalCrossUp = not na(priceLine[1]) and not na(signalLine[1]) and priceLine[1] <= signalLine[1] and priceLine > signalLine
    bool priceSignalCrossDown = not na(priceLine[1]) and not na(signalLine[1]) and priceLine[1] >= signalLine[1] and priceLine < signalLine

    [rsi, upperBand, lowerBand, marketBaseLine, priceLine, signalLine, pacUpper, pacLower, atrValue, macdLine, confirmationSignedStrength, alertSignedStrength, bandPosition, pacDistanceAtr, pacWidthAtr, bandWidthPct, bandWidthPercentile, regimeCode, riskCode, priceSignalCrossUp, priceSignalCrossDown]

// ============================================================================
// ENGINE B - FATIGUE   (aktif grafik + 8 MTF context)
// ============================================================================
f_fatigue(_tfSec) =>
    float _rsi = ta.rsi(close, fatRsiLen)
    float _rsiC = na(_rsi) ? na : math.max(0.0, math.min(100.0, _rsi))

    float _rsiS = ta.rsi(close, fatSrsiRsiLen)
    float _stRaw = ta.stoch(_rsiS, _rsiS, _rsiS, fatSrsiStochLen)
    float _k = ta.sma(_stRaw, fatSrsiKLen)
    float _kC = na(_k) ? na : math.max(0.0, math.min(100.0, _k))

    float _basis = ta.sma(close, fatBbLen)
    float _sd = ta.stdev(close, fatBbLen)
    float _up = _basis + fatBbMult * _sd
    float _lo = _basis - fatBbMult * _sd
    float _rng = _up - _lo
    float _pbRaw = na(_rng) or _rng <= 0 ? na : (close - _lo) / _rng
    float _pbSc = na(_pbRaw) ? na : math.max(-25.0, math.min(125.0, 100.0 * _pbRaw))

    bool _ok = not na(_rsiC) and not na(_kC) and not na(_pbSc)
    float _absRaw = _ok ? fatWRsi * _rsiC + fatWSrsi * _kC + fatWBb * _pbSc : na
    float _abs = na(_absRaw) ? na : math.max(0.0, math.min(100.0, _absRaw))

    int _expMs = int(_tfSec * 1000.0)
    int _delta = time - time[1]
    bool _gap = fatGapCheck and _expMs > 0 and not na(_delta) and _delta != _expMs

    var int _run = 0
    if na(_abs)
        _run := 0
    else if _gap
        _run := 1
    else
        _run := nz(_run[1]) + 1

    bool _ready = fatStrict ? _run >= fatRankWin : _run >= 1
    int _scan = fatStrict ? (_run >= fatRankWin ? fatRankWin : 0) : math.min(_run, fatRankWin)

    bool _inCalc = not limitCalc or bar_index >= last_bar_index - calcBars
    float _rankRaw = na
    if _inCalc
        _rankRaw := f_rankOf(_abs, _scan)
    float _rank = _ready ? _rankRaw : na

    float _ada = na(_abs) or na(_rank) ? na : fatWAbs * _abs + fatWRank * _rank

    var float _fat = na
    if na(_ada)
        _fat := na
    else if na(_fat[1])
        _fat := _ada
    else
        _fat := fatAlpha * _ada + (1.0 - fatAlpha) * _fat[1]
    float _fatPrev = _fat[1]

    [_fat, _fatPrev, _abs, _rank, _ada, _run]

// ============================================================================
// MTF PACK - sadece fatigue: 2 alan x 8 TF = 16 tuple elemani
// ============================================================================
f_mtfFat(_tfSec) =>
    [yFat, yFatPrev, yAbs, yRank, yAda, yRun] = f_fatigue(_tfSec)
    float vFat = yFat
    float vFatPrev = yFatPrev
    int vRun = yRun
    bool useConf = mtfMode == "CONFIRMED"
    [useConf ? vFatPrev : vFat, useConf ? vRun[1] : vRun]

f_fatRow(_t, _row, _tf, _isLtf, _fat, _run, _sz) =>
    bool ok = not _isLtf and not na(_fat)
    color fc = ok ? f_fatColor(_fat) : color.new(color.gray, 35)
    string fatTxt = _isLtf ? "LTF" : ok ? f_num(_fat, "#.##") : "-"
    string stateTxt = _isLtf ? "-" : ok ? f_fatText(_fat) : "WARMUP"
    color stateCol = _isLtf ? color.orange : fc

    bool hasRun = not _isLtf and not na(_run)
    bool ready = hasRun and (fatStrict ? _run >= fatRankWin : _run >= 1)
    string winTxt = _isLtf ? "-" : not hasRun ? "NO DATA" : str.tostring(_run, "#") + "/" + str.tostring(fatRankWin)
    color winCol = _isLtf ? color.gray : not hasRun ? color.gray : ready ? color.new(#43a047, 0) : color.new(#fb8c00, 0)

    table.cell(_t, 0, _row, f_tfLabel(_tf), text_color=color.white, text_size=_sz)
    table.cell(_t, 1, _row, fatTxt, text_color=_isLtf ? color.orange : fc, text_size=_sz, text_halign=text.align_right)
    table.cell(_t, 2, _row, stateTxt, text_color=stateCol, text_size=_sz)
    if showWindowCol
        table.cell(_t, 3, _row, winTxt, text_color=winCol, text_size=_sz)

var table mtfTbl = table.new(f_pos(mtfPos), 4, 9, border_width=1, frame_width=1, frame_color=color.new(color.gray, 50), bgcolor=color.new(color.black, 15))
