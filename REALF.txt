f_clamp(float x, float lo, float hi) =>
    math.max(lo, math.min(hi, x))

f_tanh(float x) =>
    float result = na
    if not na(x)
        float e = math.exp(2.0 * f_clamp(x, -20.0, 20.0))
        result := (e - 1.0) / (e + 1.0)
    result

f_div(float numerator, float denominator) =>
    float result = na
    if not na(numerator) and not na(denominator) and denominator > 0.0
        result := numerator / denominator
    result

// Zero is a legitimate flat-series value; missing observations stay na.
f_standard(float difference, float deviation) =>
    float result = na
    if not na(difference) and not na(deviation)
        result := 0.0
        if deviation > 1e-12
            result := difference / deviation
    result

// Origin-preserving normalization: persistent positive pressure stays positive.
// Centering net flow around its own mean would erase sustained accumulation.
f_normalize(float value, float historicalRms, float floorValue) =>
    f_tanh(f_div(value, math.max(historicalRms, floorValue)))

// Intrabar up/down classification, including doji tick direction.
f_intrabar_delta() =>
    var float direction = 0.0
    if close > open
        direction := 1.0
    else if close < open
        direction := -1.0
    else if close > close[1]
        direction := 1.0
    else if close < close[1]
        direction := -1.0
    direction * volume

// Adapter returns signed volume plus a diagnostic source flag.
// CLV fallback allocates volume by closing position in the candle range,
// instead of assigning the entire parent bar's volume to its candle color.
f_delta() =>
    float pressure = 0.0
    if high > low
        pressure := f_clamp((2.0 * close - high - low) / (high - low), -1.0, 1.0)
    float signedVol = pressure * volume
    bool usedIntrabars = false
    float contextSeconds = timeframe.in_seconds()
    string lowerTf = "1"
    if contextSeconds >= 86400
        lowerTf := "60"
    else if contextSeconds >= 14400
        lowerTf := "15"
    else if contextSeconds >= 3600
        lowerTf := "5"
    else if contextSeconds == 60 and deltaMode == "AUTO_SECONDS"
        lowerTf := "10S"
    // The comparison is evaluated INSIDE each requested TF's own context.
    // No seconds-feed subscription is required. No nested main_period grouping.
    if deltaMode != "PROXY" and (contextSeconds > 60 or (contextSeconds == 60 and deltaMode == "AUTO_SECONDS"))
        [deltas, volumes] = request.security_lower_tf("", lowerTf, [f_intrabar_delta(), volume], ignore_invalid_symbol=true, ignore_invalid_timeframe=true, calc_bars_count=60000)
        if array.size(deltas) > 0
            float intrabarVolume = array.sum(volumes)
            float intrabarDelta = array.sum(deltas)
            float coverage = f_div(intrabarVolume, volume)
            // Reject incomplete/mismatched coverage; never treat missing data as zero.
            if not na(intrabarDelta) and not na(coverage) and coverage >= 0.95 and coverage <= 1.05
                signedVol := f_clamp(intrabarDelta, -volume, volume)
                usedIntrabars := true
    [signedVol, usedIntrabars]

// Recent 100 observations plus ALL older observations at reduced weight.
// This is a moment-preserving block summary, not point subsampling.
// Old weights sum to 200; weighted mass 300 is NOT statistical effective N.
// Deeper windows cross-fade over 100 additional valid observations.
f_mr(float x) =>
    float recent = math.sum(x, 100)
    float old200 = math.sum(x[100], 200)
    float old400 = math.sum(x[100], 400)
    float old900 = math.sum(x[100], 900)
    float old2000 = math.sum(x[100], 2000)
    float observations = ta.cum(not na(x) ? 1.0 : 0.0)
    float result = (recent + old200) / 300.0
    float c500 = (recent + old400 * 0.5) / 300.0
    float c1000 = (recent + old900 * (200.0 / 900.0)) / 300.0
    float c2100 = (recent + old2000 * 0.1) / 300.0
    if not na(c500)
        result += f_clamp((observations - 500.0) / 100.0, 0.0, 1.0) * (c500 - result)
    if not na(c1000)
        result += f_clamp((observations - 1000.0) / 100.0, 0.0, 1.0) * (c1000 - result)
    if not na(c2100)
        result += f_clamp((observations - 2100.0) / 100.0, 0.0, 1.0) * (c2100 - result)
    result

// Source adapter -> engine. Supply and whale never enter the score formula.
f_engine(float signedVol, bool intrabarFlag) =>
    bool quoteUnits = volumeMode == "QUOTE" or (volumeMode == "AUTO" and syminfo.volumetype == "quote")
    bool unitsOk = volumeMode != "AUTO" or syminfo.volumetype == "base" or syminfo.volumetype == "quote"
    float notional = na
    float signedNotional = na
    if unitsOk
        notional := quoteUnits ? volume : volume * hlc3
        signedNotional := quoteUnits ? signedVol : signedVol * hlc3
    bool barOk = unitsOk and not na(volume) and volume >= 0.0 and not na(signedVol) and close > 0.0
    float validCount = math.sum(barOk ? 1.0 : 0.0, LEN)
    float baseline = f_mr(notional)
    float seedBaseline = ta.sma(notional, 20)
    float volumeBase = nz(baseline[1], seedBaseline[1])
    float rvol = f_div(notional, volumeBase)
    float pressure = volume > 0.0 ? f_clamp(signedVol / volume, -1.0, 1.0) : 0.0
    // Log activity + cap limits single anomalous volume bars.
    float flow = pressure * math.min(math.log(1.0 + rvol), 3.0)
    float ret = close > 0.0 and close[1] > 0.0 ? math.log(close / close[1]) : na
    // All regression moments use exactly the same valid observations/weights.
    float fitReturn = not na(flow) ? ret : na
    float meanR = f_mr(fitReturn)
    float meanF = f_mr(flow)
    float meanRF = f_mr(fitReturn * flow)
    float meanF2 = f_mr(flow * flow)
    float meanR2 = f_mr(fitReturn * fitReturn)
    float varianceF = math.max(meanF2 - meanF * meanF, 0.0)
    float varianceR = math.max(meanR2 - meanR * meanR, 0.0)
    float covariance = meanRF - meanR * meanF
    float betaRaw = na
    float fitR2 = na
    if not na(meanRF)
        betaRaw := varianceF > 1e-12 ? covariance / varianceF : 0.0
        fitR2 := varianceF > 1e-12 and varianceR > 1e-16 ? f_clamp(covariance * covariance / (varianceF * varianceR), 0.0, 1.0) : 0.0
    // Negative relation is diagnostic, not positive evidence of causal impact.
    float fitTrust = betaRaw[1] > 0.0 ? math.sqrt(nz(fitR2[1], 0.0)) : 0.0
    float expectedStep = betaRaw[1] * flow
    // Genuine historical beta_i * flow_i in EVERY horizon.
    float expected5 = math.sum(expectedStep, 5)
    float expected20 = math.sum(expectedStep, 20)
    float expected100 = math.sum(expectedStep, 100)
    float expected300 = math.sum(expectedStep, 300)
    float actual5 = math.sum(ret, 5)
    float actual20 = math.sum(ret, 20)
    float actual100 = math.sum(ret, 100)
    float atr = ta.atr(14)
    float priceScale = math.max(math.sqrt(meanR2[1]), math.max(0.15 * atr[1] / close[1], 1e-6))
    float flowScale = math.max(math.sqrt(meanF2[1]), 0.35)
    // Regularize the lagged coefficient, not the sign of the final score.
    // Gating the entire score by R2 can rank a strongly supported pump ABOVE
    // a weakly supported pump just because the former creates correlation.
    float fallbackBeta = priceScale / flowScale
    float calibration = fitTrust * math.max(nz(betaRaw[1], 0.0), 0.0) + (1.0 - fitTrust) * fallbackBeta
    float potentialStep = calibration * flow
    float potential5 = math.sum(potentialStep, 5)
    float potential20 = math.sum(potentialStep, 20)
    float potential100 = math.sum(potentialStep, 100)
    float gap5 = f_div(actual5 - potential5, priceScale * math.sqrt(5.0) * impactK)
    float gap20 = f_div(actual20 - potential20, priceScale * math.sqrt(20.0) * impactK)
    float gap100 = f_div(actual100 - potential100, priceScale * 10.0 * impactK)
    float impactGap = ta.ema(0.45 * f_tanh(gap5) + 0.35 * f_tanh(gap20) + 0.20 * f_tanh(gap100), 3)
    float impactScore = 50.0 + 49.0 * impactGap

    // Weak regression uses an explicitly heuristic volatility/flow scale.
    // This is NOT a fitted fair-price forecast or market-cap estimate.
    float decay = math.exp(math.log(0.5) / memoryHalfLife)
    var float inventory = na
    var int inventoryBars = 0
    if barOk and not na(potentialStep) and not na(ret)
        inventory := decay * nz(inventory[1], 0.0) + potentialStep - ret
        inventoryBars := nz(inventoryBars[1], 0) + 1
    else
        inventory := na
        inventoryBars := 0
    // Inventory is in log-return units. Repricing consumes the same state;
    // falling price adds unpriced buy pressure / consumes sell pressure.
    float memoryGap = -f_div(inventory, priceScale * math.sqrt(1.0 / (1.0 - decay)) * impactK)
    float net20 = math.sum(signedNotional, 20)
    float positiveCount = math.sum(signedNotional > 0.0 ? 1.0 : 0.0, 20)
    float negativeCount = math.sum(signedNotional < 0.0 ? 1.0 : 0.0, 20)
    float persistence = net20 > 0.0 ? positiveCount / 20.0 : (net20 < 0.0 ? negativeCount / 20.0 : 0.0)
    float persistFactor = 0.35 + 0.65 * persistence
    float memoryScore = 50.0 + 49.0 * f_tanh(memoryGap) * persistFactor

    // Matched units; magnitude retained. No independent direction-only z score.
    float fastPrice = actual5 / (priceScale * math.sqrt(5.0))
    float fastFlow = potential5 / (priceScale * math.sqrt(5.0))
    float fastGap = (actual5 - potential5) / (priceScale * math.sqrt(5.0) * impactK)
    float mediumGap = (actual20 - potential20) / (priceScale * math.sqrt(20.0) * impactK)
    float momentumGap = ta.ema(0.65 * f_tanh(fastGap) + 0.35 * f_tanh(mediumGap), 3)
    float momentumScore = 50.0 + 49.0 * momentumGap * persistFactor

    float percentile = ta.percentrank(notional, LEN)
    float activityConfidence = math.sqrt(f_clamp(rvol / 2.0, 0.0, 1.0) * f_clamp(percentile / 100.0, 0.0, 1.0))
    float core = (0.45 * impactScore + 0.25 * memoryScore + 0.20 * momentumScore) / 0.90
    float activityScore = 50.0 + (core - 50.0) * activityConfidence
    float score = na
    bool ready = validCount == LEN and inventoryBars >= 100 and not na(expected100) and not na(activityScore)
    if ready
        score := f_clamp(0.45 * impactScore + 0.25 * memoryScore + 0.20 * momentumScore + 0.10 * activityScore, 0.0, 100.0)
    float intrabarShare = ta.sma(intrabarFlag ? 1.0 : 0.0, LEN)
    float totalDeviation = 0.45 * math.abs(impactScore - 50.0) + 0.25 * math.abs(memoryScore - 50.0) + 0.20 * math.abs(momentumScore - 50.0)
    float agreement = totalDeviation > 1e-6 ? f_clamp(math.abs(0.9 * (core - 50.0)) / totalDeviation, 0.0, 1.0) : 1.0

    // Independent unusual-flow event, NOT identification of a whale trader.
    // Threshold distributions exclude the current observation.
    float p90 = ta.percentile_nearest_rank(notional, LEN, 90)[1]
    float p97 = ta.percentile_nearest_rank(notional, LEN, 97)[1]
    float p993 = ta.percentile_nearest_rank(notional, LEN, 99.3)[1]
    float pressure3 = f_div(math.sum(signedNotional, 3), math.sum(notional, 3))
    float move3 = math.sum(ret, 3)
    float response = f_div(move3, priceScale * math.sqrt(3.0))
    bool directionalFlow = math.abs(pressure3) >= 0.25 and pressure * pressure3 > 0.0
    bool absorption = math.abs(response) <= 0.75
    bool alignedResponse = response * pressure3 > 0.0
    int whaleLevel = 0
    if ready and directionalFlow and (absorption or alignedResponse)
        if notional >= p90 and rvol >= 1.5
            whaleLevel := 1
        if notional >= p97 and rvol >= 2.5
            whaleLevel := 2
        if notional >= p993 and rvol >= 4.0
            whaleLevel := 3
    int whaleDirection = whaleLevel > 0 ? (pressure3 > 0.0 ? 1 : -1) : 0
    [score, impactScore, memoryScore, momentumScore, activityScore, persistence, fastPrice, fastFlow, inventory, betaRaw, rvol, percentile, notional, whaleLevel, whaleDirection, absorption, fitR2, intrabarShare, agreement, unitsOk, expected300]

// Scalar transport: real scores are 0..100; negative codes are diagnostics ONLY.
// Each status is evaluated inside that timeframe, never inferred from chart bars.
f_readout(float score) =>
    float volumeSeen = ta.cum(nz(volume, 0.0))
    float validVolumeBars = math.sum(not na(volume) and volume >= 0.0 ? 1.0 : 0.0, LEN)
    float windowVolume = math.sum(nz(volume, 0.0), LEN)
    float result = score
    if na(score)
        if na(volume) or volumeSeen <= 0.0
            result := -1.0
        else if bar_index + 1 < 420
            result := -1000.0 - float(bar_index + 1)
        else if validVolumeBars < LEN or windowVolume <= 0.0
            result := -2.0
        else
            result := -3.0
    result

f_realf() =>
    [deltaValue, intrabarFlag] = f_delta()
    [score, impact, memory, momentum, activity, persistence, fastPrice, fastFlow, inventory, beta, rvol, percentile, notional, whaleLevel, whaleDirection, absorption, fitR2, intrabarShare, agreement, unitsOk, expected300] = f_engine(deltaValue, intrabarFlag)
    unitsOk ? f_readout(score) : -4.0

// Request the WHOLE engine in standard-bar context. Fetching only OHLC first
// and then applying history on Renko/HA would use the wrong bar sequence.
f_chart_values() =>
    [deltaValue, intrabarFlag] = f_delta()
    [score, impact, memory, momentum, activity, persistence, fastPrice, fastFlow, inventory, beta, rvol, percentile, notional, whaleLevel, whaleDirection, absorption, fitR2, intrabarShare, agreement, unitsOk, expected300] = f_engine(deltaValue, intrabarFlag)
    float readout = unitsOk ? f_readout(score) : -4.0
    [score, readout, readout[1], intrabarFlag, time, time_close, impact, memory, momentum, activity, persistence, fastPrice, fastFlow, inventory, beta, rvol, percentile, notional, whaleLevel, whaleDirection, absorption, fitR2, intrabarShare, agreement, expected300]

// Deliberate safe LTF policy: lower rows are disabled, never sampled with
// ordinary request.security(). Open a 1m chart to calculate all seven rows.
// HTF calls return ONE scalar and evaluate the whole engine in the target TF.
f_mtf(string tf) =>
    float result = na
    float targetSeconds = timeframe.in_seconds(tf)
    if showPanel
        if targetSeconds == chartSeconds
            result := matchingBar ? chartReadout : na
            if matchingBar and confirmedMode and (not barstate.isconfirmed or not timeAlignedChart)
                result := previousSourceReadout
        else if targetSeconds > chartSeconds
            if confirmedMode
                result := request.security(modelSymbol, tf, f_realf()[1], gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on, ignore_invalid_symbol=true, calc_bars_count=REQUEST_BARS)
            else
                result := request.security(modelSymbol, tf, f_realf(), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_off, ignore_invalid_symbol=true, calc_bars_count=REQUEST_BARS)
    result
